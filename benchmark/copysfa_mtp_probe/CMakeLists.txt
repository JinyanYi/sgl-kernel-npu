project(copy_refresh_candidate LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
if(NOT DEFINED SOC_VERSION)
    set(SOC_VERSION Ascend910_9382)
endif()

    get_filename_component(SRC_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)

target_link_libraries(copy_refresh_candidate PRIVATE copy_refresh_kernel torch_npu ascendcl tiling_api register platform ascendalog dl)
